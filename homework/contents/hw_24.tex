\begin{problem}
设 $W = \{ 5, 7, 10, 12, 15, 18, 20 \}$ 和 $M = 35$，使用过程 \texttt{SUMOFSUB} 找出 $W$ 中使得和数等于 $M$ 的全部子集，并画出所生成的部分状态空间树。
\end{problem}

\begin{solution}
    \begin{center}
        \begin{tikzpicture}
            \tikzset{every tree node/.style={ draw, rectangle}}
            \Tree
            [.0,1,87
            [.5,2,82
            [.12,3,75
            [.22,4,65 [.34,5,53 ]]
            [.12,4,65 [.24,5,53 ]]
            ]
            [.5,3,75
            [.15,4,65 [.15,5,53 [.15,6,38 [.15,7,20 [.35,8,0 [.(1,0,1,0,0,0,1) ]]]]]]
            [.5,4,65 [.17,5,53 [.17,6,38 [.35,7,20 [.(1,0,0,1,0,1,0) ]]]]]
            ]
            ]
            [.0,2,82
            [.7,3,75
            [.17,4,65 ]
            [.7,4,65 [.17,5,53 [.17,6,38 [.35,7,20 [.(0,1,1,0,0,1,0) ]]]]]
            ]
            [.0,3,75
            [.10,4,65 [.0,5,53 [.15,6,38 [.15,7,20 [.35,8,0 [.(0,0,0,0,1,0,1) ]]]]]]
            [.0,4,65 ]
            ]
            ]
            ]
        \end{tikzpicture}
    \end{center}
    全部子集共四个，分别为$\{5, 10, 20\}, \{5, 12, 18\}, \{7, 10, 18\}, \{15, 20\}$
\end{solution}
