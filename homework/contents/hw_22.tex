\begin{problem}
    用启发式方法求0/1背包问题.
    \begin{align*}
        n &= 5, \\
        (w_1, w_2, w_3, w_4, w_5) &= (2, 2, 6, 5, 4), \\
        (p_1, p_2, p_3, p_4, p_5) &= (6, 3, 5, 4, 6), \\
        M &= 10.
    \end{align*}
\end{problem}
\begin{solution}
    \begin{align*}
    S^0 &= \{(0,0)\} \rightarrow (w,p) = (2,6)  \\
    &\rightarrow S_1 = \{(2,6)\} 
    \\
    S^1 &= \{(0,0), (2,6)\} \rightarrow (w,p) = (2,3) \\ 
    &\rightarrow S_2 = \{(2,3),(4,9)\} 
    \\
    S^2 &= \{(0,0),(2,6),\xout{(2,3)},(4,9)\} \rightarrow (w,p) = (6,5) \\ 
    &\rightarrow S_3 = \{(6,5),(8,11),(10,14)\} 
    \\
    S^3 &= \{(0,0),(2,6),(4,9),\xout{(6,5)},(8,11),(10,14)\} \rightarrow (w,p) = (5,4) \\
    &\rightarrow S_4 = \{(5,4),(7,10),(9,13),(13,15)\} 
    \\
    S^4 &= \{(0,0),(2,6),(4,9),\xout{(5,4)},(7,10),(8,11),(9,13)\}  \rightarrow (w,p) = (4,6) \\
    &\rightarrow S_5 = \{(4,6),(6,12),(8,15),(11,16),(12,17),(13,19)\} 
    \\
    S^5 &= \{(0,0),(2,6),(4,9),\xout{(4,6)},(6,12),(7,10),\xout{(8,11)},(8,15),(9,13)\}
\end{align*}
    容易得到最大价值为15, 选取的物品为1, 2, 5.  
\end{solution}
